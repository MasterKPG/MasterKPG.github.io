\entry{\project{fir-filter}{Sequential FIR filter} (systolic project), VHDL and Quartus}{ENSSAT, second year}{Oct 2026, in progress}
\begin{itemize}
  \item 13-tap FIR filter around one multiply-accumulate unit reused 13 times: address generator,
        circular buffer, coefficient ROM, Moore FSM, testbench with a reference model. Area versus
        throughput.
\end{itemize}

\entry{\project{saturating-accumulator}{Saturating accumulator on FPGA}, VHDL}{ENSSAT, Digital Systems track}{May 2026}
\begin{itemize}
  \item 3-bit inputs, 4-bit sum saturating at 15 with an overflow flag, driven by a 5-state machine.
  \item Verified with 10 test cases in ModelSim, then on a DE10-Lite board; fixed two integration
        defects invisible in simulation (I/O standard on one pin, reset polarity).
\end{itemize}

\entry{\project{arduino-clone}{Arduino Uno clone} and \project{ping-pong}{Pong game}}{ENSSAT, class project}{Apr 2026}

\entry{\project{gate-controller}{Gate controller}, embedded software in C}{ENSSAT}{Jan 2026}
